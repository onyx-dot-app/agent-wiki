    COMMIT["7 · ✓ Commit lands"]:::ok

    L1 --> L2
    L2 -- "yes" --> L3
    L2 -- "no" --> L5
    L3 -- "conflict" --> L4

    L4 -- "agent / ingest" --> AIMERGE
    AIMERGE --> L5
    L5 --> L6
    L6 -. "beaten · retry (max 3×)" .-> L1
    L6 -- "unchanged" --> COMMIT
